private String interpretBasicCode(String code) {
    StringBuilder result = new StringBuilder();
    String[] lines = code.split("\n");

    for (String line : lines) {
        line = line.trim();
        // Skip comments, preprocessor, empty lines, etc.
        if (line.isEmpty() || line.startsWith("//") || line.startsWith("#") || line.startsWith("/*")) continue;
        if (line.startsWith("#include") || line.startsWith("using namespace") ||
            line.startsWith("public class") || line.startsWith("public static void") ||
            line.equals("{") || line.equals("}") || line.equals("return 0;")) continue;

        // 1. Try known patterns (C++, C, Java, Python, JS)
        String output = null;
        if (line.contains("cout")) {
            output = line.replaceAll(".*cout\\s*<<\\s*", "")
                         .replaceAll("\\s*<<.*", "")
                         .replaceAll("\"", "").replaceAll(";", "").trim();
        } else if (line.contains("printf")) {
            output = extractFirstQuoted(line);
        } else if (line.contains("System.out.println") || line.contains("System.out.print")) {
            output = extractFirstQuoted(line);
        } else if (line.contains("print(")) {
            output = extractFirstQuoted(line);
        } else if (line.contains("console.log")) {
            output = extractFirstQuoted(line);
        }
        // 2. Generic fallback: any quoted string in the line (works for Ruby, Go, C#, etc.)
        if (output == null || output.isEmpty()) {
            output = extractFirstQuoted(line);
        }

        if (output != null && !output.isEmpty()) {
            result.append(output).append("\n");
        }
    }
    return result.length() == 0 ? "Program finished with no output." : result.toString().trim();
}

private String extractFirstQuoted(String line) {
    int start = line.indexOf('"');
    int end = line.lastIndexOf('"');
    if (start != -1 && end > start) {
        return line.substring(start + 1, end);
    }
    return "";
}
